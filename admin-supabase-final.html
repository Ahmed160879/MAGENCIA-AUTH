-- ============================================================ -- NégoceID — Bloquer la vérification des salariés d'une -- entreprise désactivée (statut != 'actif') chez le négociant. -- -- Aujourd'hui, get_salarie_by_code() renvoie le salarié quel que -- soit le statut de son entreprise. On ajoute une jointure sur -- entreprises (via le préfixe du code, ex: "MAG-582341" -> "MAG") -- pour ne renvoyer le salarié QUE si son entreprise est "actif". -- -- Résultat : si tu désactives une entreprise dans l'admin, ses -- salariés apparaîtront comme "Non référencé" / REFUSÉ chez le -- négociant, exactement comme un code inconnu. -- -- À exécuter dans Supabase → SQL Editor → New query → Run -- ============================================================ create or replace function get_salarie_by_code(p_code text) returns table(code text, nom text, poste text, type text, statut text, chantiers text[]) language sql stable security definer as $$ select s.code, s.nom, s.poste, s.type, s.statut, s.chantiers from salaries s join entreprises e on e.prefix = split_part(s.code, '-', 1) where s.code = p_code and e.statut = 'actif' limit 1; $$; -- Vérification rapide après exécution : -- select code, nom from salaries where code = 'MAG-582341'; -- doit exister -- select get_salarie_by_code('MAG-582341'); -- doit renvoyer la ligne si l'entreprise MAG est "actif"
